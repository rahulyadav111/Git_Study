DECLARE @cols NVARCHAR(MAX);
DECLARE @sql  NVARCHAR(MAX);

-- Step 1: Build column list dynamically from PunchDates
SELECT @cols = STRING_AGG(QUOTENAME(CONVERT(varchar(10), PunchDate, 120)), ',')
FROM (SELECT DISTINCT PunchDate FROM #FinalResult WHERE PunchDate IS NOT NULL) AS d;

-- Step 2: Build the dynamic pivot query with a TotalOT column
SET @sql = '
SELECT p.Empcode, ' + @cols + ', t.TotalOT
FROM
(
    SELECT Empcode,
           CONVERT(varchar(10), PunchDate, 120) AS PunchDate,
           TotalOT
    FROM #FinalResult
    WHERE PunchDate IS NOT NULL
) src
PIVOT
(
    MAX(TotalOT) FOR PunchDate IN (' + @cols + ')
) p
INNER JOIN
(
    SELECT Empcode,
           RIGHT(''0'' + CAST(SUM(CAST(LEFT(TotalOT,2) AS INT)*60 + CAST(RIGHT(TotalOT,2) AS INT))/60 AS varchar),3)
           + '':''
           + RIGHT(''0'' + CAST(SUM(CAST(LEFT(TotalOT,2) AS INT)*60 + CAST(RIGHT(TotalOT,2) AS INT))%60 AS varchar),2) AS TotalOT
    FROM #FinalResult
    WHERE ISDATE(TotalOT) = 1
    GROUP BY Empcode
) t
ON p.Empcode = t.Empcode
ORDER BY p.Empcode;
';
